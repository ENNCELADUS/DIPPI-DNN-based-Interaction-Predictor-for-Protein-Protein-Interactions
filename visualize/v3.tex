% Inputs
\coordinate (inputmid) at (0,0);
\node[input, yshift=1.8cm] (protA) at (inputmid) {\vlabel{Protein A Input\\$m$}};
\node[input, yshift=-1.8cm] (protB) at (inputmid) {\vlabel{Protein B Input\\$n$}};

% ESM backbone
\node[esm, right=1.6cm of inputmid] (esm) {\vlabel{ESM-3}};

% Embeddings
\node[embedA, right=1.0cm of esm, yshift=1.8cm] (embA) {\vlabel{A Embedding\\$m \times 1536$}};
\node[embedB, right=1.0cm of esm, yshift=-1.8cm] (embB) {\vlabel{B Embedding\\$n \times 1536$}};

% Siamese encoder (shared weights)
\node[transformer, right=1.0cm of embA] (trA) {\vlabel{MHSA Transformer\\$1536 \rightarrow 512$}};
\node[transformer, right=1.0cm of embB] (trB) {\vlabel{MHSA Transformer\\$1536 \rightarrow 512$}};

\node[embedA, right=1.0cm of trA] (encA) {\vlabel{A Encoded\\$m \times 512$}};
\node[embedB, right=1.0cm of trB] (encB) {\vlabel{B Encoded\\$n \times 512$}};

% Cross-attention and concat
\node[crossattn, right=1.0cm of encA] (xattA) {\vlabel{Cross-Attention\\(A attends to B)}};
\node[crossattn, right=1.0cm of encB] (xattB) {\vlabel{Cross-Attention\\(B attends to A)}};

\coordinate (xattmid) at ($(xattA)!0.5!(xattB)$);
\node[concat, right=1.5cm of xattmid] (concat) {\vlabel{A concat B\\$(m+n) \times 512$}};

% CLS cross-attention
\node[cls, above=1.0cm of concat] (initcls) {\vlabel{Init CLS\\$1 \times 512$}};
\node[crossattn, right=1.0cm of concat] (clsattn) {\vlabel{Cross-Attention\\(CLS attends to concat)}};
\node[ffn, right=1.0cm of clsattn] (clsffn) {\vlabel{CLS FFN\\(Linear-GELU-Linear)}};
\node[cls, right=1.0cm of clsffn] (clsout) {\vlabel{CLS\\$1 \times 512$}};

% MLP head and output
\node[head, right=1.0cm of clsout] (mlp1) {\vlabel{Linear + LN + GELU\\Dropout 0.2\\$512 \rightarrow 256$}};
\node[head, right=1.0cm of mlp1] (mlp2) {\vlabel{Linear + LN + GELU\\Dropout 0.2\\$256 \rightarrow 128$}};
\node[head, right=1.0cm of mlp2] (mlp3) {\vlabel{Linear\\$128 \rightarrow 1$}};
\node[output, right=1.0cm of mlp3] (score) {\vlabel{Score\\$1$}};
